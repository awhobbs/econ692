\section{The Introduction Formula}
% ══════════════════════════════════════════════════════════

\begin{frame}{The Introduction Formula}
  Whether it's a paper or a blog post, the intro follows the same structure:

  \vspace{8pt}

  \begin{enumerate}
    \item \textbf{Motivation} --- a fact, cost, or puzzle that creates stakes
    \item \textbf{Research question} --- ``I estimate\ldots'' or ``This project examines\ldots''
    \item \textbf{Contribution} --- what's new, positioned against prior work
    \item \textbf{Method} --- your identification strategy, in one paragraph
    \item \textbf{Findings} --- main results with actual numbers/directions
    \item \textbf{Roadmap} --- section outline (keep it short)
  \end{enumerate}

  \vspace{8pt}

  {\scriptsize Based on Evans (CGDev), Head (UBC), and Sahm (MacroMom). Links on course site.}
\end{frame}

\begin{frame}{The First Three Paragraphs}
  \footnotesize
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{1. Motivation}}
      \begin{itemize}\setlength\itemsep{0pt}
        \item It \emph{matters} --- people are helped or hurt
        \item It's \emph{puzzling} --- defies easy explanation
        \item It's \emph{controversial} --- economists disagree
        \item It's \emph{big} or widespread
      \end{itemize}
      Be specific: a number, a policy, a cost.

      \vspace{3pt}

      \textbf{\textcolor{usfgreen}{2. Research Question}}
      \begin{itemize}\setlength\itemsep{0pt}
        \item By end of \P 2: ``I estimate\ldots'' or ``This project examines\ldots''
        \item After citing others, use ``I'' or ``my analysis'' --- ``this'' is ambiguous
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{3. Contribution}}
      \begin{itemize}\setlength\itemsep{0pt}
        \item Aim for $\sim$3 contributions
        \item Most important first
        \item New economic knowledge > new method/data
        \item Frame each against specific prior work
      \end{itemize}

      \vspace{3pt}

      \textbf{\textcolor{usfgreen}{The test:}} if a reader stops after \P 3, do they know (1) why the topic matters, (2) what you did, and (3) why it's new?
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Method, Findings, Roadmap}
  \footnotesize
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{4. Method (1 paragraph)}}
      \begin{itemize}\setlength\itemsep{0pt}
        \item Summarize your identification strategy
        \item Name the design: DiD, IV, RDD, etc.
        \item Save details for the empirical strategy section
      \end{itemize}

      \vspace{4pt}

      \textbf{\textcolor{usfgreen}{5. Findings (1--2 paragraphs)}}
      \begin{itemize}\setlength\itemsep{0pt}
        \item One sentence per main result --- with direction and magnitude
        \item ``I find that a 10\% increase in X reduces Y by 3\%''
        \item Don't make the reader wait 20 pages to find out what you found
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{6. Roadmap (short)}}
      \begin{itemize}\setlength\itemsep{0pt}
        \item One sentence per section
        \item Be specific to \emph{your} project
        \item Not: ``The next section reviews the literature''
        \item Better: ``Section 2 documents trends in rural hospital closures since 2010''
      \end{itemize}

      \vspace{4pt}

      \textbf{Total length:} 1.5--2 pages for a capstone, 3--5 paragraphs for a blog post. The intro is an executive summary --- it should stand alone.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Lead with Your Work}
  Every element of the intro should open with \emph{your} contribution, then position it in the literature.

  \vspace{8pt}

  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \textbf{\textcolor{red!60}{Don't:}}

      \vspace{2pt}

      ``Smith (2010) found X. Jones (2012) extended this to Y. Lee (2015) showed Z. Building on this literature, I also examine\ldots''

      \vspace{4pt}

      {\scriptsize Three sentences about other people before the reader finds out what \emph{you} did.}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{\textcolor{usfgreen}{Do:}}

      \vspace{2pt}

      ``I provide the first causal estimates of X using staggered policy variation. While Smith (2010) and Jones (2012) documented correlations, my design exploits\ldots''

      \vspace{4pt}

      {\scriptsize Your contribution is the first thing the reader sees. Literature supports it.}
    \end{column}
  \end{columns}

  \vspace{8pt}

  This applies to every paragraph: contribution, method, findings. Lead with yours, then cite. And don't trash prior work --- tell us what \emph{your} analysis adds.
\end{frame}

\begin{frame}{Common Intro Mistakes}
  \begin{itemize}
    \item \textbf{Too broad.} ``Since the beginning of time, economies have\ldots'' Start with your specific topic.

    \vspace{3pt}

    \item \textbf{Bait and switch.} The hook promises something interesting, but the analysis delivers something narrower and less compelling.

    \vspace{3pt}

    \item \textbf{``All my friends are doing it.''} Motivating your project because others wrote on the topic, not because the question itself matters.

    \vspace{3pt}

    \item \textbf{Burying the contribution.} If your reader gets to page 3 without knowing what you did, you've lost them.

    \vspace{3pt}

    \item \textbf{No results preview.} The intro should tell the reader what you \emph{found}. An intro without findings is a promissory note.

    \vspace{3pt}

    \item \textbf{Mixing intro and lit review.} Cite a few key papers to position your work. Save the deep engagement for the body.
  \end{itemize}
\end{frame}

\section{Optional Sections}
% ══════════════════════════════════════════════════════════

\begin{frame}{Do You Need a Background Section?}
  \small
  Some papers include a standalone section between the intro and empirical strategy. You don't have to --- but here's when it helps:

  \vspace{4pt}

  \begin{itemize}\setlength\itemsep{2pt}
    \item \textbf{Institutional detail.} Your setting needs explanation that doesn't fit in the intro --- how a policy works, a timeline of changes, features of a market.
    \item \textbf{Literature review.} The intro cites a few key papers. A lit review goes deeper --- grouping by theme, synthesizing findings, identifying gaps.
    \item \textbf{Theory / conceptual framework.} A model or predictions that motivate your empirical tests. Common in IO and trade, less so in applied micro.
  \end{itemize}

  \vspace{4pt}

  If you include one, it has one job: make the reader understand \emph{why your empirical strategy makes sense}. Otherwise, fold the key points into your intro.
\end{frame}

\section{Writing the Data Section}
% ══════════════════════════════════════════════════════════

\begin{frame}{The Data Section}
  The data section answers: \textbf{what are you working with, and should I trust it?}

  \vspace{6pt}

  \begin{enumerate}
    \item \textbf{Source.} Where does the data come from? How was it collected?
    \item \textbf{Unit of observation.} What is a row? Person-year? County-month? Firm?
    \item \textbf{Sample.} Time period, geographic scope, sample size. What's excluded and why?
    \item \textbf{Key variables.} How are treatment, outcome, and controls measured? Be precise --- ``income'' could mean ten different things.
    \item \textbf{Summary statistics.} A table of means, SDs, and counts. Separate by treatment/control if relevant.
  \end{enumerate}

  \vspace{6pt}

  A reader who skips straight to your results table should be able to come back here and understand every column.
\end{frame}

\section{Anatomy of the Empirical Strategy}
% ══════════════════════════════════════════════════════════

\begin{frame}{The Empirical Strategy Recipe}
  This section should convince a skeptic that your approach is sound.

  \vspace{8pt}

  Four ingredients, in order:

  \vspace{4pt}

  \begin{enumerate}
    \item \textbf{Data.} Unit of observation, time period, sample size, key variables. Where does it come from? What's excluded and why?
    \item \textbf{Identification.} What variation are you exploiting? Why is it plausibly exogenous? What's the identifying assumption?
    \item \textbf{Estimation.} The model. Write down the equation. Explain every term.
    \item \textbf{Inference.} How are standard errors clustered? Why that level?
  \end{enumerate}

  \vspace{6pt}

  If you don't have an equation in this section, something is missing.
\end{frame}

\begin{frame}{The Identification Argument}
  The identification argument is the heart of your empirical strategy. It answers: \textbf{why should I believe this is causal?}

  \vspace{6pt}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{For DiD:}}
      \begin{itemize}
        \item What's the treatment? When did it happen?
        \item Who's treated vs.\ control?
        \item Why is parallel trends plausible?
        \item What threatens it?
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{For IV:}}
      \begin{itemize}
        \item What's the instrument?
        \item Why is it relevant? (first stage)
        \item Why is it excludable?
        \item What would violate exclusion?
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{8pt}

  For every method: state the assumption, argue for it, and acknowledge what could go wrong.
\end{frame}

\begin{frame}{Common Empirical Strategy Mistakes}
  \begin{itemize}
    \item \textbf{No equation.} ``I run a regression of Y on X with controls'' isn't enough. Write it out:
    \[Y_{it} = \alpha + \beta D_{it} + X_{it}'\gamma + \mu_i + \lambda_t + \varepsilon_{it}\]

    \vspace{4pt}

    \item \textbf{Skipping the ``why this works'' argument.} Don't just describe what you do. Explain \emph{why} it gives you a credible estimate.

    \vspace{4pt}

    \item \textbf{Mixing in results.} This section is about \emph{design}, not findings. Save the coefficients for the results section.

    \vspace{4pt}

    \item \textbf{No discussion of threats.} Every strategy has weaknesses. Name them. The reader will think of them anyway --- better if you get there first.
  \end{itemize}
\end{frame}

\section{Results, Discussion, Conclusion}
% ══════════════════════════════════════════════════════════

\begin{frame}{The Results Section}
  We covered this in detail last week. The short version:

  \vspace{6pt}

  \begin{enumerate}
    \item \textbf{Lead with your main result.} State the finding in plain English before the reader hits the table.
    \item \textbf{Walk through the table.} Point to the key coefficient, its magnitude, and its significance. Don't just say ``see Table 2.''
    \item \textbf{Build from simple to complex.} Start with the baseline specification, then add controls, fixed effects, robustness checks.
    \item \textbf{Interpret economically, not just statistically.} ``A 10\% increase in X reduces Y by 3 percentage points'' is more useful than ``$p < 0.01$.''
  \end{enumerate}

  \vspace{6pt}

  Every table and figure should be referenced in the text. If you don't discuss it, cut it.
\end{frame}

\begin{frame}{Discussion and Limitations}
  \small
  This is where you step back and honestly assess what your results mean --- and what they don't.

  \vspace{6pt}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{Discussion}}
      \begin{itemize}\setlength\itemsep{0pt}
        \item What do the results imply for policy or theory?
        \item Do they match or contradict prior work? Why?
        \item What's the mechanism? Can you say?
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{Limitations}}
      \begin{itemize}\setlength\itemsep{0pt}
        \item What threats to identification remain?
        \item What can't your data measure?
        \item How does sample selection affect generalizability?
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{6pt}

  Be honest but not self-defeating. Name the limitations, explain what you did to address them, and note what future work could do. Don't undermine your own results --- just show you've thought it through.
\end{frame}

\begin{frame}{The Conclusion}
  The conclusion is short --- half a page to a page. Three things:

  \vspace{8pt}

  \begin{enumerate}
    \item \textbf{Restate the question and answer.} One sentence each. The reader who skipped to the end should get the punchline.

    \vspace{4pt}

    \item \textbf{So what?} Why does this matter? Policy implications, theoretical implications, or both.

    \vspace{4pt}

    \item \textbf{What's next?} One or two sentences on future work --- what would make this analysis stronger or extend it in an interesting direction?
  \end{enumerate}

  \vspace{8pt}

  Don't introduce new results or new arguments. Don't repeat the abstract. Don't end with ``more research is needed'' --- be specific about \emph{what} research and \emph{why}.
\end{frame}

\section{Good vs.\ Bad Examples}
% ══════════════════════════════════════════════════════════

\begin{frame}{Intro: Weak vs.\ Strong}
  \small
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{red!60}{Weak:}}

      \vspace{2pt}

      ``Minimum wages have been debated for a long time. Many economists have studied this topic. Some find positive effects and some find negative effects. I look at minimum wages and employment.''

      \vspace{4pt}

      {\scriptsize \textcolor{darkgray}{Problems: no hook, no stakes, no specific question, vague contribution.}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{Strong:}}

      \vspace{2pt}

      ``Twenty-three states raised their minimum wage in 2024, affecting over 8 million workers. Whether these increases reduce employment remains contested. I exploit the staggered timing of state-level increases to estimate the effect on teen employment using a stacked DiD design.''

      \vspace{4pt}

      {\scriptsize \textcolor{darkgray}{Hook (fact), stakes (scale), question (implicit), method (clear).}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Empirical Strategy: Weak vs.\ Strong}
  \footnotesize
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{red!60}{Weak:}}

      \vspace{2pt}

      ``I use difference-in-differences. The treatment group is states that raised the minimum wage. The control group is states that didn't. I control for GDP and population.''

      \vspace{4pt}

      {\scriptsize \textcolor{darkgray}{No equation, no identifying assumption, no parallel trends discussion.}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{usfgreen}{Strong:}}

      \vspace{2pt}

      ``I estimate $Y_{st} = \beta D_{st} + X_{st}'\gamma + \mu_s + \lambda_t + \varepsilon_{st}$, where $D_{st}$ indicates state $s$ raised its minimum wage by period $t$. Identification requires parallel employment trends absent the policy. I provide evidence in Section~5.''

      \vspace{4pt}

      {\scriptsize \textcolor{darkgray}{Equation, named assumption, roadmap to evidence.}}
    \end{column}
  \end{columns}
\end{frame}

\section{GitHub Pages}
% ══════════════════════════════════════════════════════════

\begin{frame}{Blog Series: The First Post Matters Most}
  If you're writing a blog series, your first post is your intro --- and it needs to be a full executive summary, not just ``I'm going to study X.''

  \vspace{6pt}

  A first post that doesn't tell the reader what you found won't get them to post two. Include:
  \begin{itemize}
    \item Why this question matters (the hook)
    \item What you did and what you found (the punchline)
    \item What's coming in the rest of the series (the roadmap)
  \end{itemize}

  \vspace{6pt}

  The remaining sections --- data, empirical strategy, results, discussion --- each become their own post. Same content, more conversational tone.

  \vspace{6pt}

  Let's talk about how to set that up with GitHub Pages.
\end{frame}

\begin{frame}{What Is GitHub Pages?}
  Free static website hosting, straight from a GitHub repo.

  \vspace{6pt}

  \begin{itemize}
    \item Your repo already has code and data. GitHub Pages lets you add a website on top of it.
    \item URL: \texttt{yourusername.github.io/your-repo/}
    \item Supports HTML, Markdown, and --- most usefully --- Quarto.
    \item No server to manage. Push your files, the site updates.
  \end{itemize}

  \vspace{8pt}

  \begin{block}{Two Options}
    \begin{enumerate}
      \item \textbf{Simple:} Write Markdown files, enable Pages, done.
      \item \textbf{Polished:} Use Quarto to build a blog or multi-page site.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}[fragile]{Option 1: Simple Markdown Site}
  \small
  If you just want a single-page write-up:

  \vspace{2pt}

  \begin{enumerate}\setlength\itemsep{0pt}
    \item Create \texttt{docs/index.md} in your repo
    \item Write your content in Markdown
    \item \textbf{Settings} $\to$ \textbf{Pages} $\to$ source: \texttt{main} branch, \texttt{/docs} folder
    \item Push. Your site is live.
  \end{enumerate}

  \vspace{4pt}

  \begin{lstlisting}[style=terminal,basicstyle=\ttfamily\scriptsize]
your-repo/
  docs/
    index.md      # your write-up
    figures/       # images for the site
  code/
    analysis.R
  \end{lstlisting}

  \vspace{2pt}

  GitHub renders the Markdown as HTML automatically. No build step needed.
\end{frame}

\begin{frame}[fragile]{Option 2: Quarto Blog}
  \small
  For multiple posts, a polished layout, or interactive figures:

  \vspace{2pt}

  \begin{lstlisting}[style=terminal,basicstyle=\ttfamily\scriptsize]
# Create a blog project
quarto create project blog my-capstone-blog

# Structure:
my-capstone-blog/
  _quarto.yml           # site config
  index.qmd             # landing page
  posts/
    01-question/
      index.qmd         # "Why I'm studying this"
    02-data/
      index.qmd         # "The data"
    03-results/
      index.qmd         # "What I found"
  \end{lstlisting}

  \vspace{2pt}

  Run \texttt{quarto render} to build, then push the output to GitHub Pages.
\end{frame}

\begin{frame}[fragile]{Setting Up GitHub Pages}
  \small
  \begin{enumerate}\setlength\itemsep{1pt}
    \item Go to your repo on GitHub
    \item \textbf{Settings} $\to$ \textbf{Pages} (left sidebar)
    \item Source: \textbf{Deploy from a branch}, branch: \texttt{main}, folder: \texttt{/docs}
    \item Click \textbf{Save}
  \end{enumerate}

  \vspace{4pt}

  \textbf{For Quarto:} set the output directory in \texttt{\_quarto.yml}:

  \begin{lstlisting}[style=terminal,basicstyle=\ttfamily\scriptsize]
project:
  type: website
  output-dir: docs
  \end{lstlisting}

  \vspace{2pt}

  Then \texttt{quarto render \&\& git add docs/ \&\& git push}. Site updates in about a minute.
\end{frame}

\section{Peer Feedback \& Wrap-up}
% ══════════════════════════════════════════════════════════

\begin{frame}{Peer Feedback (7:30--7:55)}
  Swap your intro (or first blog post) with a neighbor.

  \vspace{8pt}

  \textbf{5 minutes:} Read silently. Then answer:

  \vspace{4pt}

  \begin{enumerate}
    \item Can you state their research question in one sentence?
    \item Is the contribution clear --- do you know what they did and what they found?
    \item Where did you get confused or lose the thread?
  \end{enumerate}

  \vspace{8pt}

  If you can't answer \#1 after reading the intro, that's the most useful feedback you can give.
\end{frame}
